    <script>
        // 전역 변수 선언
        let map = null;          // 지도 객체
        let ps = null;           // 장소 검색 서비스 객체
        let markers = [];        // 지도 위에 표시된 마커들을 담아둘 배열
        let currentPosition = null; // 사용자의 현재 위치 좌표

        // 페이지가 완전히 로드된 후 카카오맵 SDK 로드 시작
        window.addEventListener('DOMContentLoaded', () => {
            // kakao.maps.load()를 사용하여 지도가 완전히 준비된 후에 실행되도록 보장
            kakao.maps.load(() => {
                initMap();
            });
        });

        /**
         * 1. 지도 초기화 및 위치 권한 요청 함수
         */
        function initMap() {
            const mapContainer = document.getElementById('map'); // 지도를 표시할 div
            
            // 기본 중심 좌표 (위치 권한 거부 또는 오류 시 보여줄 기본 위치: 서울 시청)
            let defaultLat = 37.566826;
            let defaultLng = 126.978656;

            const mapOption = {
                center: new kakao.maps.LatLng(defaultLat, defaultLng), // 지도의 중심좌표
                level: 5 // 지도 확대 레벨 (요청 조건: 5)
            };

            // 지도 생성
            map = new kakao.maps.Map(mapContainer, mapOption);
            
            // 장소 검색 객체 생성 (카카오 맵 라이브러리)
            ps = new kakao.maps.services.Places();

            // 브라우저 Geolocation API를 이용해 현재 위치 권한 요청
            if (navigator.geolocation) {
                navigator.geolocation.getCurrentPosition(
                    (position) => {
                        // 위치 권한 허용 성공 시
                        const lat = position.coords.latitude; // 위도
                        const lng = position.coords.longitude; // 경도
                        
                        currentPosition = new kakao.maps.LatLng(lat, lng);
                        
                        // 현재 위치로 지도 중심 이동
                        map.setCenter(currentPosition);
                        
                        // 현재 위치에 마커 표시
                        displayCurrentLocationMarker(currentPosition);
                    },
                    (error) => {
                        // 위치 권한 거부 또는 오류 발생 시
                        console.warn('위치 권한을 얻지 못했습니다. 기본 위치로 설정합니다.', error);
                        currentPosition = new kakao.maps.LatLng(defaultLat, defaultLng);
                    },
                    {
                        enableHighAccuracy: true,
                        timeout: 5000,
                        maximumAge: 0
                    }
                );
            } else {
                console.warn('이 브라우저는 Geolocation을 지원하지 않습니다.');
                currentPosition = new kakao.maps.LatLng(defaultLat, defaultLng);
            }
        }

        /**
         * 2. 현재 위치 마커 표시 함수
         */
        function displayCurrentLocationMarker(locPosition) {
            // 현재 위치를 알리는 커스텀 마커 이미지 또는 기본 마커 생성
            const marker = new kakao.maps.Marker({
                map: map,
                position: locPosition
            });

            // 인포윈도우로 현재 위치 표시
            const iwContent = '<div style="padding:5px; font-size:12px; text-align:center; width:120px;">내 현재 위치</div>';
            const infowindow = new kakao.maps.InfoWindow({
                content: iwContent
            });

            infowindow.open(map, marker);
        }

        /**
         * 3. '내 위치' 버튼 클릭 시 현재 위치로 지도 이동하는 함수
         */
        function moveToCurrentLocation() {
            if (currentPosition) {
                map.setCenter(currentPosition);
                map.setLevel(5);
            } else {
                alert('현재 위치 정보를 아직 불러오지 못했습니다.');
            }
        }
